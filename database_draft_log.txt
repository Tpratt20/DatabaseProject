    -> RETURNS DECIMAL(10,2)
    -> READS SQL DATA
    -> BEGIN
    ->     DECLARE order_total DECIMAL(10,2);
    ->     SELECT COALESCE(SUM(Quantity * UnitPrice), 0)
    ->     INTO order_total
    -> FROM OrderItems

    ->     RETURN order_total;
